### 1.3 빌드/실행 절차

#### 1.3.1 소스 클론
```bash

cd S14P31C101
```

- Docker engine을 실행한다 
```bash

```

```bash

```


## 4. nginx 설정 
```yaml
server {
    listen 80;
    server_name api.timingnote.co.kr;

    # Certbot 인증용 경로
    location /.well-known/acme-challenge/ {
        root /var/www/certbot;
    }

    # 기본적으로 모든 HTTP 요청을 HTTPS로 리다이렉트
    location / {
        return 301 https://$host$request_uri;
    }
}

server { # -> 인증서 발급 후
    listen 443 ssl;
    server_name api.timingnote.co.kr;

    # 요청 본문 최대 크기
    client_max_body_size 20m;

    # [중요] 도커 컨테이너 내부 기준의 인증서 경로
    ssl_certificate /etc/letsencrypt/live/api.timingnote.co.kr/fullchain.pem;
    ssl_certificate_key /etc/letsencrypt/live/api.timingnote.co.kr/privkey.pem;

    # SSL 보안 설정 (보통 이 설정을 권장함)
    ssl_protocols TLSv1.2 TLSv1.3;
    ssl_prefer_server_ciphers on;
    
    # 모든 location 블록에 공통으로 적용될 명함(헤더)
    proxy_set_header Host $host;
    proxy_set_header X-Real-IP $remote_addr;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;

    # SSE 전용 (정확 경로)
    location = /api/v1/notifications/geofence/slots/stream {
        proxy_pass http://springboot:8080;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_buffering off;
        proxy_cache off;

        # 추가 권장
        proxy_set_header X-Accel-Buffering no;
        add_header Cache-Control "no-cache";
        add_header X-Accel-Buffering "no";

        # 필요하면 더 길게 (예: 2h)
        proxy_read_timeout 3600s;
        proxy_send_timeout 3600s;
    }

    # 2) 자동 스캔 타겟 경로 선차단 (추가)
    location ~* ^/(phpmyadmin|phpmyadmin4\.8\.5|pma|pmd|adminer|wp-admin|wp-login|vendor|\.env)(?:/|$) {
        return 403;   # 필요 시 444로 변경 가능
        access_log off;
        log_not_found off;
    }

    # 3) .php 요청 선차단 (추가)
    location ~* \.php(?:$|/) {
        return 403;
        access_log off;
        log_not_found off;
    }

    # / 요청은 고정 응답 반환
    location = / {
        return 200 "ok";
        add_header Content-Type text/plain always;
    }
    location = /. { return 204; }
    location = /actuator/health { return 204; }
    # 숨김 파일/경로 차단 (Let's Encrypt 제외)
    location ~ /\.(?!well-known).* {
        return 404;
        access_log off;
        log_not_found off;
    }

    location = /robots.txt { return 204; }
    location = /sitemap.xml { return 204; }
    location = /favicon.ico { return 204; }
    location = /.well-known/security.txt { return 204; }

    # jenkins 연결
    location ^~ /jenkins {
        proxy_pass http://jenkins:8080;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_buffering off;
    }

    location = /grafana {
        return 301 /grafana/;
    }

    location ^~ /grafana/ {
        proxy_pass http://grafana:3000;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
        proxy_set_header Host $host;
        proxy_set_header X-Real-IP $remote_addr;
        proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
        proxy_set_header X-Forwarded-Proto https;
        proxy_set_header X-Forwarded-Host $host;
        proxy_set_header X-Forwarded-Port 443;
    }

    

    # Spring Boot 연결
    location /api/ {
        proxy_pass http://springboot:8080;
        proxy_http_version 1.1;
        proxy_set_header Connection "";
    }

    #그 외 전부 차단
    location / {
        return 404;  # 또는 403/444
        access_log off;
        log_not_found off;
    }
    
    }
```

## 5. infra docker-compose.yml
```yaml
services:
  nginx:
    image: nginx:latest # nginx 설치
    container_name: nginx
    ports:
      - "80:80" # [호스트 포트]:[컨테이너 포트]
      - "443:443"
    volumes: # 로컬 경로와 컨테이너 경로 매핑 or 덮어쓰기 -> 로컬에서 설정 후 컨테이너에 적용
      - ./nginx/config:/etc/nginx/conf.d # [로컬 경로]:[컨테이너 내부 경로]
      - ./nginx/html:/usr/share/nginx/html
      - ./certbot/conf:/etc/letsencrypt
      - ./certbot/www:/var/www/certbot
    restart: always
    networks:
      - tn-network

  certbot:
    image: certbot/certbot # cerbot 설치
    container_name: certbot
    volumes:
      - ./certbot/conf:/etc/letsencrypt
      - ./certbot/www:/var/www/certbot
    # 인증서 자동 갱신 로직 (12시간마다 확인)
    entrypoint: "/bin/sh -c 'trap exit TERM; while :; do certbot renew; sleep 12h & wait $${!}; done;'"
    networks:
      - tn-network

  jenkins :
    build:
      context: ./jenkins
      dockerfile: Dockerfile
    container_name: jenkins
    expose:
      - "8080" #Jenkins는 외부에 노출하지 않고, Nginx 컨테이너에서 jenkins:8080으로만 접근가능하도록 설정
    volumes:
      - ./jenkins-data:/var/jenkins_home
      - /var/run/docker.sock:/var/run/docker.sock
    group_add:
    - "${DOCKER_GID}"
    environment:
      - JENKINS_OPTS=--prefix=/jenkins #Nginx 리버스 프록시 대응 설정
      - TZ=Asia/Seoul
    restart : always
    networks:
      - tn-network

networks:
  tn-network:
      name: tn-network
  
```
